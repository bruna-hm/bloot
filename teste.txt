do {
    # Solicita ao usuário que insira um IP
    $ip = Read-Host "Digite um endereço IP"

    # Tenta validar o IP usando [System.Net.IPAddress]::TryParse()
    $isValid = [System.Net.IPAddress]::TryParse($ip, [ref]$null)

    if (-not $isValid) {
        Write-Host "O endereço IP '$ip' não é válido. Tente novamente." -ForegroundColor Red
    }
} while (-not $isValid)

Write-Host "O endereço IP '$ip' é válido!" -ForegroundColor Green
